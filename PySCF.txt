import csv, os, time, py3Dmol
import cupy as cp
import numpy as np
import matplotlib.pyplot as plt
from rdkit import Chem
from rdkit.Chem import AllChem
from pyscf import gto, scf
from pyscf.geomopt.berny_solver import optimize
from pyscf.geomopt.addons import as_pyscf_method
from gpu4pyscf.dft import rks, uks

HARTREE2EV = 27.211386245988


# 1. SMILES -> 3D 좌표
def smiles_to_geometry(smiles, seed):
    """
    SMILES 문자열을 받아 (원소 기호 리스트, 좌표 배열)을 반환 (단위: Angstrom)
    내부적으로 RDKit의 ETKDGv3 embedding + MMFF94 force-field 최적화를 사용해 초기 조건으로 입력될 3D 구조를 만든다.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        raise ValueError("유효하지 않은 SMILES입니다.")
    mol = Chem.AddHs(mol)
    params = AllChem.ETKDGv3()
    params.useRandomCoords = True
    params.randomSeed = seed
    
    if AllChem.EmbedMolecule(mol, params) < 0:
        raise RuntimeError("3D 임베딩에 실패하였습니다.")
    try:
        AllChem.MMFFOptimizeMolecule(mol)
    except Exception:
        AllChem.UFFOptimizeMolecule(mol)

    conf = mol.GetConformer()
    symbols = [atom.GetSymbol() for atom in mol.GetAtoms()]
    coords = conf.GetPositions()
    return symbols, coords


def build_pyscf_atom_string(symbols, coords):
    lines = []
    for s, (x, y, z) in zip(symbols, coords):
        lines.append(f"{s} {x:.6f} {y:.6f} {z:.6f}")
    return "\n".join(lines)


def run_single_molecule(name, smiles, basis='6-31G', method='b3lyp', charge=0, spin=0, maxsteps=30, seed=42, outdir='.', show_view=True):
    os.makedirs(outdir, exist_ok=True)
    
    # 2. 분자 객체 생성 및 SMILES로부터 초기 구조 설정
    symbols, init_coords = smiles_to_geometry(smiles, seed=seed)
    atom_str = build_pyscf_atom_string(symbols, init_coords)
    print(f"\n{'='*60}\n[{name}] SMILES = {smiles}\nbasis set = {basis}\nmethod={method}\n{'='*60}")
    print(f"원자 수: {len(symbols)}개")
    
    # 3. 계산 방법 설정 및 optimization step data 저장 설정
    out_log_file = f"{name}_opt.log"
    basis_dict = {
        'Cu': 'lanl2dz',
        'C': '6-31G',
        'H': '6-31G',
        'N': '6-31G',
        'O': '6-31G'
    }
    ecp_dict = {'Cu': 'lanl2dz'}
    mol = gto.M(atom=atom_str,
                basis=basis_dict,
                ecp=ecp_dict,
                unit="Angstrom",
                charge=charge,
                spin=spin,
                verbose=4,
                output=out_log_file,
                max_memory=10000)

    
    def scan_fn(mol):
        mf = rks.RKS(mol, xc=method) if spin == 0 else uks.UKS(mol, xc=method)
        mf.grids.level = 0
        mf.conv_tol=1e-6
        e = mf.kernel()
        if isinstance(e, cp.ndarray):
            e = float(e.get())
        else:
            e = float(e)
        g = mf.nuc_grad_method().kernel()
        if isinstance(g, cp.ndarray):
            g = g.get()
        else:
            g = np.asarray(g)
        return e, g

    history = {"energies": [], "geometries": [], "atom_symbols": None}
    
    
    def record_step(envs):
        current_mol = envs["mol"]
        history["energies"].append(envs["energy"])
        history["geometries"].append(current_mol.atom_coords(unit="Angstrom").copy())
        if history["atom_symbols"] is None:
            history["atom_symbols"] = [current_mol.atom_symbol(i) for i in range(current_mol.natm)]
    
    # 4. Geometry optimization 수행
    print("--- 구조 최적화 시작 ---")
    start_time = time.perf_counter()
    mol_eq = optimize(as_pyscf_method(mol, scan_fn), callback=record_step, maxsteps=maxsteps)
    elapsed_sec = int(time.perf_counter() - start_time)
    
    n_steps = len(history["energies"])
    symbols = history["atom_symbols"]
    final_geo = history["geometries"][-1]
    final_energy = history['energies'][-1] * HARTREE2EV
    print(f"\n--- Geometry Optimization 완료 ({n_steps} step, {elapsed_sec}초 소요) ---")
    print(f"최종 에너지: {final_energy:.4f} eV")

    xyz_path = f"{name}_opt.xyz"
    with open(xyz_path, 'w') as f:
        f.write(f"{len(symbols)}\n")
        f.write(f"{name} ({smiles}), final energy = {final_energy:.4f} eV\n")
        for s, (x, y, z) in zip(symbols, final_geo):
            f.write(f"{s} {x:.6f} {y:.6f} {z:.6f}\n")
    
    # 5. Energy plotting
    energies = np.array(history["energies"])
    relative_energies = (energies - energies[-1]) * HARTREE2EV
    steps = np.arange(1, n_steps+1)
    
    plt.figure()
    plt.plot(steps, relative_energies, marker="o", linestyle="-", color="red", linewidth=2)
    plt.title("Geometry Optimization Energy Convergence", fontsize=14, pad=12)
    plt.xticks(steps)
    plt.xlabel("Optimization Step", fontsize=12)
    plt.ylabel("Relative Energy (eV)", fontsize=12)
    plt.grid(True, linestyle="--", alpha=0.6)
    plt.tight_layout()
    plt.show()
    
    # 6. 최적화 구조 시각화
    with open(xyz_path, "r") as f:
        xyz_data = f.read()
    view = py3Dmol.view(width=400, height=300)
    view.addModelsAsFrames(xyz_data, "xyz")
    view.setStyle({"stick": {"radius": 0.15}, "sphere": {"scale": 0.3}})
    view.zoomTo()
    if show_view:
        view.show()

    return {'name': name, 'smiles': smiles, 'basis': basis, 'method': method, 'charge': charge, 'n_steps': n_steps,
        'final_energy_eV': final_energy, 'elapsed_sec': elapsed_sec, 'xyz_path': xyz_path}
